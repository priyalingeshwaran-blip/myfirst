# Program 1: Menu driven program - Armstrong Number and Prime Number

def is_armstrong(num):
    """Check whether a number is an Armstrong number."""
    temp = num
    n = len(str(num))
    total = 0
    while temp > 0:
        digit = temp % 10
        total += digit ** n
        temp //= 10
    return total == num


def is_prime(num):
    """Check whether a number is prime."""
    if num < 2:
        return False
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            return False
    return True


def main():
    while True:
        print("\n----- MENU -----")
        print("a. Armstrong Number")
        print("b. Prime Number")
        print("c. Exit")
        choice = input("Enter your choice (a/b/c): ").lower()

        if choice == 'a':
            num = int(input("Enter a number: "))
            if is_armstrong(num):
                print(f"{num} is an Armstrong number.")
            else:
                print(f"{num} is not an Armstrong number.")

        elif choice == 'b':
            num = int(input("Enter a number: "))
            if is_prime(num):
                print(f"{num} is a Prime number.")
            else:
                print(f"{num} is not a Prime number.")

        elif choice == 'c':
            print("Exiting program...")
            break

        else:
            print("Invalid choice! Please try again.")


if __name__ == "__main__":
    main()

# Program 2: Menu driven program - Palindrome, Fibonacci series, Factorial

def is_palindrome(s):
    """Check whether a string is a palindrome."""
    return s == s[::-1]


def fibonacci_series(n):
    """Return Fibonacci series up to n terms as a list."""
    series = []
    a, b = 0, 1
    for _ in range(n):
        series.append(a)
        a, b = b, a + b
    return series


def factorial(n):
    """Return factorial of a number."""
    fact = 1
    for i in range(1, n + 1):
        fact *= i
    return fact


def main():
    while True:
        print("\n----- MENU -----")
        print("a. Palindrome of a given string")
        print("b. Fibonacci series up to n terms")
        print("c. Factorial of a given number")
        print("d. Exit")
        choice = input("Enter your choice (a/b/c/d): ").lower()

        if choice == 'a':
            s = input("Enter a string: ")
            if is_palindrome(s):
                print(f"'{s}' is a palindrome.")
            else:
                print(f"'{s}' is not a palindrome.")

        elif choice == 'b':
            n = int(input("Enter number of terms: "))
            print("Fibonacci series:", fibonacci_series(n))

        elif choice == 'c':
            n = int(input("Enter a number: "))
            print(f"Factorial of {n} is {factorial(n)}")

        elif choice == 'd':
            print("Exiting program...")
            break

        else:
            print("Invalid choice! Please try again.")


if __name__ == "__main__":

# Program 3: Print patterns (stars, numbers, letters) using functions
# Pattern 1 (stars):     *
#                        * *
#                        * * *
#                        * * * *
#                        * * * * *
#
# Pattern 2 (numbers):   54321
#                        4321
#                        321
#                        21
#                        1
#
# Pattern 3 (letters):   A
#                        BB
#                        CCC
#                        DDDD
#                        EEEEE

def star_pattern(n):
    """Return list of strings for the star pattern."""
    lines = []
    for i in range(1, n + 1):
        lines.append(" ".join(["*"] * i))
    return lines


def number_pattern(n):
    """Return list of strings for the descending number pattern."""
    lines = []
    for i in range(n, 0, -1):
        # for row size i, numbers go from i down to 1
        row = "".join(str(num) for num in range(i, 0, -1))
        lines.append(row)
    return lines


def letter_pattern(n):
    """Return list of strings for the letter pattern (A, BB, CCC, ...)."""
    lines = []
    for i in range(1, n + 1):
        letter = chr(64 + i)  # 65 = 'A'
        lines.append(letter * i)
    return lines


def print_patterns(n=5):
    stars = star_pattern(n)
    numbers = number_pattern(n)
    letters = letter_pattern(n)

    for i in range(n):
        # left-align each column for a neat side-by-side display
        print(f"{stars[i]:<20}{numbers[i]:<20}{letters[i]:<20}")


if __name__ == "__main__":
    print_patterns(5)

# Program 5: Accept N strings, store in a tuple, find longest and shortest

def accept_strings(n):
    """Accept n strings from the user and return them as a tuple."""
    strings = []
    for i in range(n):
        s = input(f"Enter string {i + 1}: ")
        strings.append(s)
    return tuple(strings)


def find_longest_shortest(string_tuple):
    """Return the longest and shortest string from the tuple."""
    longest = max(string_tuple, key=len)
    shortest = min(string_tuple, key=len)
    return longest, shortest


def main():
    n = int(input("Enter the number of strings: "))
    string_tuple = accept_strings(n)
    print("\nTuple of strings:", string_tuple)

    longest, shortest = find_longest_shortest(string_tuple)
    print("Longest string:", longest)
    print("Shortest string:", shortest)

# Program 6: Count uppercase, lowercase, vowels, consonants using a dictionary

def count_characters(s):
    """Accept a string and return counts of upper, lower, vowels, consonants."""
    counts = {"uppercase": 0, "lowercase": 0, "vowels": 0, "consonants": 0}
    vowels = "aeiouAEIOU"

    for ch in s:
        if ch.isupper():
            counts["uppercase"] += 1
        if ch.islower():
            counts["lowercase"] += 1
        if ch.isalpha():
            if ch in vowels:
                counts["vowels"] += 1
            else:
                counts["consonants"] += 1

    return counts


def main():
    s = input("Enter a string: ")
    result = count_characters(s)

    print("\nUppercase letters:", result["uppercase"])
    print("Lowercase letters:", result["lowercase"])
    print("Vowels:", result["vowels"])
    print("Consonants:", result["consonants"])

    main()

Accept a list of N elements and perform:

  1. Swap first half with second half.
  2. Multiply elements in even position by 5.

Answer:

# Program 7: Accept a list of N elements and:
# i)  Swap elements in the first half with the second half
# ii) Multiply elements in the even position with 5

def accept_list(n):
    """Accept n elements from the user and return as a list."""
    lst = []
    for i in range(n):
        val = int(input(f"Enter element {i + 1}: "))
        lst.append(val)
    return lst


def swap_halves(lst):
    """Swap the first half of the list with the second half."""
    n = len(lst)
    mid = n // 2
    # If n is odd, the middle element stays in place
    first_half = lst[:mid]
    second_half = lst[n - mid:]
    middle = lst[mid:n - mid]  # empty if n is even
    return second_half + middle + first_half


def multiply_even_position(lst):
    """Multiply elements at even positions (0-based index 1,3,5,...) by 5.
    Here 'even position' is taken as position number 2,4,6,... (i.e. index 1,3,5,...)."""
    new_list = lst.copy()
    for i in range(len(new_list)):
        position = i + 1  # 1-based position
        if position % 2 == 0:
            new_list[i] *= 5
    return new_list


def main():
    n = int(input("Enter number of elements: "))
    lst = accept_list(n)
    print("\nOriginal list:", lst)

    swapped = swap_halves(lst)
    print("List after swapping halves:", swapped)

    multiplied = multiply_even_position(swapped)
    print("List after multiplying even position elements by 5:", multiplied)


if __name__ == "__main__":
    main()

import math


def generate_three_digit_random():
    """Generate and return a random three digit number (100-999)."""

def arc_length(radius, angle_degrees):
    """Calculate arc length given radius and angle in degrees.
    Arc Length = (angle/360) * 2 * pi * radius """
    return (angle_degrees / 360) * 2 * math.pi * radius

    """Create mother.txt and write some information into it."""
    with open("mother.txt", "w") as f:
        f.write("A mother is the first teacher of a child.\n")
        f.write("She gives unconditional love and care.\n")
        f.write("Her sacrifices cannot be measured.\n")

    """Read and print the content of a file."""
    print(f"\n----- Content of {filename} -----")
    with open(filename, "r") as f:
        print(f.read())


def copy_without_letter_A():
    """Read mother.txt line by line and write to parent.txt,
    skipping lines that contain the letter 'A' (case-sensitive, capital A)."""
    with open("mother.txt", "r") as source, open("parent.txt", "w") as dest:
        for line in source:
            if "A" not in line:
                dest.write(line)
    print("parent.txt created successfully (lines with 'A' excluded).")


def main():
    create_mother_file()
    display_file("mother.txt")

    copy_without_letter_A()
    display_file("parent.txt")

def count_word_occurrences(filename, word):
    """Count occurrences of a specific word (case-sensitive) in the file."""
    count = 0
    with open(filename, "r") as f:
        for line in f:
            words = line.split()
            count += words.count(word)
    return count


def display_lines_starting_with(filename, letter):
    """Display all lines that start with the given letter."""
    print(f"\nLines starting with '{letter}':")
    with open(filename, "r") as f:
        for line in f:
            if line.strip().startswith(letter):
                print(line.strip())


def main():

import pickle


def insert_employees():
    """Accept employee details and append them to Employee.dat"""
    n = int(input("How many employee records do you want to enter? "))
    with open("Employee.dat", "ab") as f:
        for i in range(n):
            print(f"\nEnter details for employee {i + 1}:")
            empid = input("Empid: ")
            name = input("Name: ")
            dept = input("Department: ")
            salary = float(input("Salary: "))
            record = {"Empid": empid, "Name": name,
                      "Department": dept, "Salary": salary}
            pickle.dump(record, f)
    print("Records inserted successfully.")


def update_salary(increment=5000, threshold=15000):
    """Update all records where salary > threshold by adding increment."""
    records = []
    try:
        with open("Employee.dat", "rb") as f:
            while True:
                try:
                    record = pickle.load(f)
                    if record["Salary"] > threshold:
                        record["Salary"] += increment
                    records.append(record)
                except EOFError:
                    break
    except FileNotFoundError:
        print("Employee.dat not found.")
        return

    with open("Employee.dat", "wb") as f:
        for record in records:
            pickle.dump(record, f)
    print(f"Salaries updated for records with salary > {threshold}.")


def display_and_count():
    """Read Employee.dat, display all records and count them."""
    count = 0
    try:
        with open("Employee.dat", "rb") as f:
            print("\n----- Employee Records -----")
            while True:
                try:
                    record = pickle.load(f)
                    print(record)
                    count += 1
                except EOFError:
                    break
    except FileNotFoundError:
        print("Employee.dat not found.")
        return
    print(f"\nTotal number of records: {count}")


def main():
    while True:
        print("\n----- MENU -----")
        print("1. Insert employee records")
        print("2. Update salary (salary > 15000, increment 5000)")
        print("3. Display all records and count")
        print("4. Exit")
        choice = input("Enter your choice: ")

        if choice == '1':
            insert_employees()
        elif choice == '2':
            update_salary()
        elif choice == '3':
            display_and_count()
        elif choice == '4':
            break
        else:
            print("Invalid choice!")

    main()

# Program 16: Implement a stack using a list

stack = []


def push(item):
    """Push an item onto the stack."""
    stack.append(item)
    print(f"{item} pushed to stack.")


def pop():
    """Pop and return the top item of the stack."""
    if not stack:
        print("Stack Underflow! Stack is empty.")
        return None
    item = stack.pop()
    print(f"{item} popped from stack.")
    return item


def peek():
    """Return the top item of the stack without removing it."""
    if not stack:
        print("Stack is empty.")
        return None
    print(f"Top element is: {stack[-1]}")
    return stack[-1]


def display():
    """Display the current stack."""
    if not stack:
        print("Stack is empty.")
    else:
        print("Stack (top to bottom):", stack[::-1])


def main():
    while True:
        print("\n----- STACK MENU -----")
        print("1. Push")
        print("2. Pop")
        print("3. Peek")
        print("4. Display")
        print("5. Exit")
        choice = input("Enter your choice: ")

        if choice == '1':
            item = input("Enter item to push: ")
            push(item)
        elif choice == '2':
            pop()
        elif choice == '3':
            peek()
        elif choice == '4':
            display()
        elif choice == '5':
            print("Exiting program...")
            break
        else:
            print("Invalid choice!")



    main()
